🧠 2. Evaluación exacta (core del sistema)
✅ Ejemplo con decimal.js
import Decimal from "decimal.js";

function evaluateExpression(numbers: string[]) {
  return numbers.reduce(
    (acc, n) => acc.plus(new Decimal(n)),
    new Decimal(0)
  );
}

✔ Resultado exacto
✔ Sin errores flotantes

🧠 3. Alternativa PRO: math.js con BigNumber
import { create, all } from "mathjs";

const math = create(all);
math.config({ number: "BigNumber", precision: 64 });

const result = math.evaluate("0.1 + 0.2");

✔ Precisión arbitraria
✔ Soporta expresiones complejas

🧪 5. Validación completa de pregunta
function validateMathQuestion(question) {
  const result = evaluate(question.expresion);

  const correctOptions = question.opciones.filter(opt =>
    result.equals(new Decimal(opt))
  );

  if (correctOptions.length === 0) {
    return { valid: false, reason: "No correct answer" };
  }

  if (correctOptions.length > 1) {
    return { valid: false, reason: "Multiple correct answers" };
  }

  return { valid: true };
}

export const QuestionSchema = z.object({
  pregunta: z.string(),
  expresion: z.string().optional(),
  opciones: z.array(z.string()).length(4),
  respuesta_correcta: z.string(),
  explicacion: z.string(),

🧮 VALIDACIÓN MATEMÁTICA

Usa:

decimal.js
math.js
Flujo
expresión → parser → BigNumber → resultado exacto → comparación
Ejemplo
import { create, all } from "mathjs";

const math = create(all);
math.config({ number: "BigNumber", precision: 64 });

function validateMath(q) {
  const result = math.evaluate(q.expresion);

  const matches = q.opciones.filter(opt =>
    math.equal(result, math.bignumber(opt))
  );

  if (matches.length !== 1) {
    throw new Error("Invalid math question");
  }

  return true;
}
